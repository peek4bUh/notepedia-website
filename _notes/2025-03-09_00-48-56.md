# Getting Help

```bash
# Display the man page for a command
man command

# List all available sections for a command
man -f command

# Display the man page for a command from
# specific section listed above
man section_number command

# Display the path of manpages
# OR display the location of a command's manpage
man -w
man -w command

# Search for manpages containing that keyword
# OR display a description from a command's manpage
man -k "keyword"
whatis command
whatis -w wildcard
whatis -r regex

# Search the `PATH` environment variable and display
# the location of any matching executables
# OR If there are multiple executables which match, display all
which command
which -a <𝗘𝗫𝗘𝗖𝗨𝗧𝗔𝗕𝗟𝗘>

# Search for a keyword using regex
# OR Search for a keyword using regex (Long Output)
apropos regex
apropos -l regex

# Search for pages that match all the expressions given
apropos regex1 -a regex2 -a regex3...

# Full list of builtin commands
help

# Print instructions on how to use that builtin command
help builtin_command

# Display the type of a builtin command
type builtin_command

# Display the type of multiple builtin commands
type builtin_command1 builtin_command2...

# Display all locations of a builtin command
type -a builtin_command
```
