# Enable Debian Backports on RPI

```bash
# Verify what release you are (e.g, bullseye, bookworm, etc...)
cat /etc/apt/sources.list*

# Install Debian Archive Keyring (the deb package can be updated wit time)
curl -O http://http.us.debian.org/debian/pool/main/d/debian-archive-keyring/debian-archive-keyring2023.4all.deb

# Install it
sudo dpkg -i debian-archive-keyring2023.4all.deb

# Create a new file with the new backports repo
# (you can add the echo line inside /etc/apt/sources.list file too)
echo "deb http://deb.debian.org/debian bookworm-backports main" \
  | sudo tee /etc/apt/sources.list.d/bookworm-backports.list
```
